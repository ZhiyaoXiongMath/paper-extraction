% Rendered-size budgets. U is the fixed normal body baseline at document start.
% Blocks are typeset ONCE in a scratch vbox, measured, then unboxed unchanged.
% No clipping, scaling, automatic truncation, or hidden duplicate text is used.
\newdimen\IntroUnit
\newdimen\IntroHeight
\newdimen\IntroLimit
\newdimen\IntroGroupHeight
\newdimen\IntroProseHeight
\newdimen\IntroSharedHeight
\newbox\IntroMeasuredBox
\newcount\IntroBlockCount
\newcount\IntroGroupID
\newcount\IntroActiveGroup
\newcount\IntroGroupTheorems
\newcount\IntroGroupDefinitions
\newcount\IntroSharedDefinitions
\newcount\IntroProseCount
\AtBeginDocument{\global\IntroUnit=\baselineskip
  \typeout{INTRO-UNIT-SP: \number\IntroUnit}}
\newcommand{\IntroBudgetError}[1]{%
  \PackageError{intro-v13}{#1}{Rewrite the selected mathematics. Do not change the budget or font.}}
\newcommand{\IntroMeasured}[3]{%
  \par
  \global\advance\IntroBlockCount by 1
  \setbox\IntroMeasuredBox=\vbox{\hsize=\linewidth #3\par}%
  \IntroHeight=\dimexpr\ht\IntroMeasuredBox+\dp\IntroMeasuredBox\relax
  \IntroLimit=#2\IntroUnit
  \typeout{INTRO-BLOCK|\the\IntroBlockCount|#1|\number\IntroHeight|\number\IntroLimit|\the\IntroActiveGroup}%
  \ifdim\IntroHeight>\IntroLimit
    \IntroBudgetError{Block \the\IntroBlockCount\space (#1) exceeds #2U}%
  \fi
  \ifnum\IntroActiveGroup>0
    \global\advance\IntroGroupHeight by \IntroHeight
  \fi
  \ifstrequal{#1}{P}{\global\advance\IntroProseHeight by \IntroHeight}{}%
  \ifstrequal{#1}{D}{\ifnum\IntroActiveGroup=0
    \global\advance\IntroSharedHeight by \IntroHeight\fi}{}%
  % Each measured block is bounded: preserve it as one readable unit.
  \Needspace{\dimexpr\IntroHeight+4pt\relax}%
  \unvbox\IntroMeasuredBox
  \par\addvspace{4pt}%
}
\NewDocumentEnvironment{introparagraph}{+b}{%
  \global\advance\IntroProseCount by 1
  \IntroMeasured{P}{8}{\noindent #1}%
}{}
\NewDocumentEnvironment{definition}{O{} +b}{%
  \ifnum\IntroActiveGroup>0
    \global\advance\IntroGroupDefinitions by 1
    \ifnum\IntroGroupDefinitions>2 \IntroBudgetError{More than two dedicated definitions in one result group}\fi
  \else
    \global\advance\IntroSharedDefinitions by 1
    \ifnum\IntroSharedDefinitions>2 \IntroBudgetError{More than two shared definitions}\fi
  \fi
  \IntroMeasured{D}{10}{\begin{rawdefinition}[{#1}]#2\end{rawdefinition}}%
}{}
\NewDocumentEnvironment{historical}{O{} +b}{%
  \ifnum\value{rawtheorem}>0 \IntroBudgetError{Historical theorem placed after current results}\fi
  \ifnum\IntroActiveGroup>0 \IntroBudgetError{History is not part of a current-result group}\fi
  \IntroMeasured{H}{11}{\begin{rawhistorical}[{#1}]#2\end{rawhistorical}}%
}{}
\NewDocumentEnvironment{theorem}{O{} +b}{%
  \ifnum\IntroActiveGroup=0 \IntroBudgetError{Current theorem requires a resultgroup}\fi
  \global\advance\IntroGroupTheorems by 1
  \IntroMeasured{T}{16}{\begin{rawtheorem}[{#1}]#2\end{rawtheorem}}%
}{}
\newenvironment{resultgroup}{%
  \ifnum\IntroActiveGroup>0 \IntroBudgetError{Nested resultgroup}\fi
  \global\advance\IntroGroupID by 1
  \global\IntroActiveGroup=\IntroGroupID
  \global\IntroGroupHeight=0pt
  \global\IntroGroupTheorems=0
  \global\IntroGroupDefinitions=0
}{%
  \IntroLimit=32\IntroUnit
  \typeout{INTRO-GROUP|\the\IntroGroupID|\number\IntroGroupHeight|\number\IntroLimit|\the\IntroGroupTheorems|\the\IntroGroupDefinitions}%
  \ifdim\IntroGroupHeight>\IntroLimit \IntroBudgetError{Result group exceeds 32U}\fi
  \ifnum\IntroGroupTheorems=1\else \IntroBudgetError{A result group must contain exactly one theorem}\fi
  \global\IntroActiveGroup=0
}
% Independent hypotheses: every source row begins with &, so their left edges align.

% A genuine equality/inequality chain: exactly one relation anchor in each row.
\NewDocumentEnvironment{relations}{+b}{\[\begin{aligned}#1\end{aligned}\]}{}
\AtEndDocument{%
  \typeout{INTRO-COUNTS|\arabic{rawhistorical}|\arabic{rawtheorem}|\arabic{rawdefinition}|\the\IntroProseCount|\the\IntroGroupID}%
  \ifnum\value{rawhistorical}>3 \IntroBudgetError{More than three historical theorems}\fi
  \ifnum\value{rawtheorem}<1 \IntroBudgetError{At least one admitted current theorem is required}\fi
  % The current-theorem count is selected under the user-authorized scope.
  \ifnum\value{rawdefinition}>6 \IntroBudgetError{More than six definitions}\fi
  \IntroLimit=40\IntroUnit
  \ifdim\IntroProseHeight>\IntroLimit \IntroBudgetError{Total connecting prose exceeds 40U}\fi
  \IntroLimit=20\IntroUnit
  \ifdim\IntroSharedHeight>\IntroLimit \IntroBudgetError{Shared definitions exceed 20U}\fi
  \ifnum\IntroActiveGroup>0 \IntroBudgetError{Unclosed result group}\fi
}

    {Supply all verified source authors in source order.}}{}%
